    // Cálculo geométrico del volumen firmado de triángulos

    function computeMeshVolume(geometry) {

      const pos = geometry.attributes.position;

      if (!pos) return 15000;

      let vol = 0;

      const p1 = new THREE.Vector3(), p2 = new THREE.Vector3(), p3 = new THREE.Vector3();

      const index = geometry.index;

 

      if (index) {

        for (let i = 0; i < index.count; i += 3) {

          p1.fromBufferAttribute(pos, index.getX(i));

          p2.fromBufferAttribute(pos, index.getX(i + 1));

          p3.fromBufferAttribute(pos, index.getX(i + 2));

          vol += p1.dot(p2.cross(p3)) / 6.0;

        }

      } else {

        for (let i = 0; i < pos.count; i += 3) {

          p1.fromBufferAttribute(pos, i);

          p2.fromBufferAttribute(pos, i + 1);

          p3.fromBufferAttribute(pos, i + 2);

          vol += p1.dot(p2.cross(p3)) / 6.0;

        }

      }

      return Math.abs(vol);

    }

    // Parser ligero para archivos STL (Binario y ASCII) y OBJ reales subidos por el usuario

    function parseUploadedFile(file) {

      const ext = file.name.split('.').pop().toLowerCase();

      const reader = new FileReader();

 

      if (ext === 'stl') {

        reader.onload = function (event) {

          try {

            const buffer = event.target.result;

            const geometry = parseSTLBuffer(buffer);

            displayGeometryInViewer(geometry, file.name);

            showFileBanner(`Archivo "${file.name}" analizado con éxito.`);

          } catch (err) {

            showFileBanner(`Error al leer "${file.name}". Verifica que sea un STL válido.`);

          }

        };

        reader.readAsArrayBuffer(file);

      } else if (ext === 'obj') {

        reader.onload = function (event) {

          try {

            const text = event.target.result;

            const geometry = parseOBJText(text);

            displayGeometryInViewer(geometry, file.name);

            showFileBanner(`Archivo OBJ "${file.name}" analizado con éxito.`);

          } catch (err) {

            showFileBanner(`Error al leer "${file.name}". Verifica el formato OBJ.`);

          }

        };

        reader.readAsText(file);

      } else {

        showFileBanner('Formato no soportado en el visor. Sube un archivo .STL o .OBJ.');

      }

    }

 

    function parseSTLBuffer(buffer) {

      const dataView = new DataView(buffer);

      const isBinary = buffer.byteLength > 84 && (84 + dataView.getUint32(80, true) * 50 === buffer.byteLength);

 

      const vertices = [];

      if (isBinary) {

        const faces = dataView.getUint32(80, true);

        let offset = 84;

        for (let i = 0; i < faces; i++) {

          offset += 12; // saltar normal

          for (let v = 0; v < 3; v++) {

            vertices.push(

              dataView.getFloat32(offset, true),

              dataView.getFloat32(offset + 4, true),

              dataView.getFloat32(offset + 8, true)

            );

            offset += 12;

          }

          offset += 2; // byte count

        }

      } else {

        const text = new TextDecoder().decode(buffer);

        const vertexRegex = /vertex\s+([+-]?\d*\.?\d+(?:[eE][+-]?\d+)?)\s+([+-]?\d*\.?\d+(?:[eE][+-]?\d+)?)\s+([+-]?\d*\.?\d+(?:[eE][+-]?\d+)?)/g;

        let match;

        while ((match = vertexRegex.exec(text)) !== null) {

          vertices.push(parseFloat(match[1]), parseFloat(match[2]), parseFloat(match[3]));

        }

      }

 

      const geometry = new THREE.BufferGeometry();

      geometry.setAttribute('position', new THREE.Float32BufferAttribute(vertices, 3));

      return geometry;

    }

 

    function parseOBJText(text) {

      const verts = [];

      const triVerts = [];

      const lines = text.split('\n');

      for (const line of lines) {

        const parts = line.trim().split(/\s+/);

        if (parts[0] === 'v' && parts.length >= 4) {

          verts.push([parseFloat(parts[1]), parseFloat(parts[2]), parseFloat(parts[3])]);

        } else if (parts[0] === 'f' && parts.length >= 4) {

          const faceIndices = parts.slice(1).map(p => parseInt(p.split('/')[0], 10) - 1);

          for (let i = 1; i < faceIndices.length - 1; i++) {

            const a = verts[faceIndices[0]];

            const b = verts[faceIndices[i]];

            const c = verts[faceIndices[i + 1]];

            if (a && b && c) {

              triVerts.push(...a, ...b, ...c);

            }

          }

        }

      }

      const geometry = new THREE.BufferGeometry();

      geometry.setAttribute('position', new THREE.Float32BufferAttribute(triVerts, 3));

      return geometry;

    }

    // Recálculo dinámico de precio, peso, entrega y enlace pre-rellenado de WhatsApp

    function recalculateQuote() {

      const effectiveFillFactor = 0.35 + (quoteState.infillPercent / 100) * 0.65;

      const singleWeightG = quoteState.volumeCm3 * quoteState.materialDensity * effectiveFillFactor;

      const totalWeightG = singleWeightG * quoteState.quantity;

 

      // Precio base por unidad con tarifa mínima de preparación de máquina

      const setupFee = 80;

      const unitMaterialCost = quoteState.volumeCm3 * quoteState.materialRate * quoteState.infillMult * quoteState.layerMult;

      let subtotal = setupFee + (unitMaterialCost * quoteState.quantity);

 

      // Descuento progresivo por volumen (series cortas)

      let discount = 0;

      if (quoteState.quantity >= 50) discount = 0.22;

      else if (quoteState.quantity >= 15) discount = 0.14;

      else if (quoteState.quantity >= 5) discount = 0.08;

 

      const discountBadge = document.getElementById('discount-badge');

      if (discount > 0) {

        discountBadge.textContent = `-${Math.round(discount * 100)}% por lote`;

        discountBadge.classList.remove('hidden');

      } else {

        discountBadge.classList.add('hidden');

      }

 

      const finalPrice = Math.round(subtotal * (1 - discount));

 

      // Tiempo estimado

      let deliveryText = "24 - 48 horas";

      if (quoteState.quantity >= 50) deliveryText = "4 - 6 días hábiles";

      else if (quoteState.quantity >= 10) deliveryText = "2 - 4 días hábiles";

 

      document.getElementById('est-weight').textContent = totalWeightG >= 1000

        ? `${(totalWeightG / 1000).toFixed(2)} kg`

        : `${totalWeightG.toFixed(1)} g`;

      document.getElementById('est-delivery').textContent = deliveryText;

      document.getElementById('est-price').textContent = `$${finalPrice.toLocaleString('es-MX')}`;

 

      // Construir mensaje detallado para WhatsApp

      const waMessage = [

        `Hola *PrintMakers 3D*, me gustaría cotizar y producir la siguiente pieza:`,

        ``,

        `📁 *Archivo:* ${quoteState.fileName}`,

        `📐 *Dimensiones:* ${quoteState.dimX} × ${quoteState.dimY} × ${quoteState.dimZ} mm (${quoteState.volumeCm3} cm³)`,

        `🧪 *Material:* ${quoteState.materialName} (${quoteState.materialTech})`,

        `🧱 *Relleno (Infill):* ${quoteState.infillPercent}%`,

        `🎯 *Calidad de Capa:* ${quoteState.layerLabel}`,

        `🔢 *Cantidad:* ${quoteState.quantity} pza(s)`,

        `💰 *Estimado Web:* $${finalPrice.toLocaleString('es-MX')} MXN`,

        ``,

        `Adjunto mi archivo a continuación para validación técnica.`

      ].join('\n');

 

      const waUrl = `https://wa.me/${WHATSAPP_NUMBER}?text=${encodeURIComponent(waMessage)}`;

      document.getElementById('send-quote-whatsapp').setAttribute('href', waUrl);

    }
